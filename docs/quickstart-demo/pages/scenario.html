import { defineDemo } from '@plaintake/scenario'; export default defineDemo({  schema: 'agent-demo.scenario/v1',  id: 'release-approval', title: 'Approve the 0.2.0 release',  viewport: { width: 1920, height: 1080 },}); await demo.step({  id: 'enter-version', title: 'Enter the version',  subtitle: 'Name the version being released.',  target: page.getByLabel('Version'), action: 'type',  run: () => page.getByLabel('Version').fill('0.2.0'),}); await demo.assert({  id: 'approval-visible',  run: () => page.getByRole('heading', { name: 'Release 0.2.0 is approved.' }).waitFor(),});
